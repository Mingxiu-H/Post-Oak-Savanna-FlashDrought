// Load MODIS AET / PET data
var et = ee.ImageCollection("MODIS/061/MOD16A2GF")
            .select(['ET','PET'])
            .filterDate('2000-01-01', '2024-12-31')
            .filterBounds(POS);

// Compute ESR = ET / PET for each grid cell
//Convert from kg m⁻² per 8 days → mm per 8 days 
//(multiply by 0.1 if needed—MODIS stores ET × 0.1 mm).
var esr = et.map(function(img){
  var et_mm  = img.select('ET').multiply(0.1);
  var pet_mm = img.select('PET').multiply(0.1);
  return et_mm.divide(pet_mm)
              .updateMask(pet_mm.gt(0))
              .rename('ESR')
              .copyProperties(img, ['system:time_start']);
});

// 8-day SESR standardization (per bin across all years)

var STEP_DAYS = 8;
var NBINS     = 46; // ~365/8
var bins = ee.List.sequence(0, NBINS - 1);

// Tag each ESR image with its 8-day-of-year bin (0..45)
function addBin(img){
  var t   = ee.Date(img.get('system:time_start'));
  var y0  = ee.Date.fromYMD(t.get('year'), 1, 1);
  var doy = t.difference(y0, 'day');              // 0..364/365
  var bin = doy.divide(STEP_DAYS).floor().int();  // 0..45
  return img.set('bin', bin);
}

// Tag bins directly on un-clipped ESR
var esr_b = esr.map(addBin);

// use the precomputed ESR climatology
// Put the exact asset path you see in the Assets panel:
var baseProj = et.first().select('ET').projection();
var CLIM = ee.Image('users/mingxiuw/POS_ESR_climatology_bins_2024')
  .setDefaultProjection(baseProj);


// SESR(t) = (ESR - mean_bin) / std_bin using stored climatology
// For each image at bin p:
var sesr = esr_b.map(function(img){
  var bstr = ee.Number(img.get('bin')).format('%02d');  // '00'..'45'
  var meanName = ee.String('mean_bin_').cat(bstr);
  var stdName  = ee.String('std_bin_').cat(bstr);

  var mu  = CLIM.select(meanName);
  var sig = CLIM.select(stdName);

  return img.subtract(mu).divide(sig)
           .updateMask(sig.gt(0))
           .rename('SESR')
           .copyProperties(img, ['system:time_start'])
           .set('bin', img.get('bin'));
});

// 2. Compute ΔSESR = SESR(t) − SESR(t−1), allowing cross-year differences
var dsesr = ee.ImageCollection(
  ee.List.sequence(1, sesrList.size().subtract(1)).map(function(i) {
    i = ee.Number(i);
    var curr = ee.Image(sesrList.get(i));
    var prev = ee.Image(sesrList.get(i.subtract(1)));

    var diff = curr.subtract(prev)
                   .rename('dSESR')
                   .copyProperties(curr, curr.propertyNames());

    return diff;
  })
);


// 3. Bin-based ΔSESR climatology: mean and std per bin
var dMeanIC = ee.ImageCollection(
  bins.map(function(b){
    b = ee.Number(b);
    var coll = dsesr.filter(ee.Filter.eq('bin', b));
    return coll.mean().rename('tmp');
  })
).toBands();

var dMeanNames = bins.map(function(b){
  return ee.String('dmean_').cat(ee.Number(b).format('%02d'));
});
var dsesrMean = dMeanIC.rename(dMeanNames);

var dStdIC = ee.ImageCollection(
  bins.map(function(b){
    b = ee.Number(b);
    var coll = dsesr.filter(ee.Filter.eq('bin', b));
    return coll.reduce(ee.Reducer.stdDev()).rename('tmp');
  })
).toBands();

var dStdNames = bins.map(function(b){
  return ee.String('dstd_').cat(ee.Number(b).format('%02d'));
});
var dsesrStd = dStdIC.rename(dStdNames);

// 4. Standardized ΔSESR (SdSESR) for each time step
//    SdSESR = (dSESR - mean_bin) / std_bin
//    For criterion 4, later we'll use mean(SdSESR) < -0.674
var SdSESR = dsesr.map(function(img){
  var bstr = ee.Number(img.get('bin')).format('%02d');

  var meanName = ee.String('dmean_').cat(bstr);
  var stdName  = ee.String('dstd_').cat(bstr);

  var mu  = dsesrMean.select(meanName);
  var sig = dsesrStd.select(stdName);

  return img.subtract(mu)
            .divide(sig)
            .updateMask(sig.gt(0))
            .rename('SdSESR')
            .copyProperties(img, ['system:time_start', 'bin']);
});

var sesrSorted  = sesrWithP20.sort('system:time_start');
var dsesrSorted = dsesrWithP40.sort('system:time_start');

function dsesrPercentileByBin(p) {
  var pIC = ee.ImageCollection(
    bins.map(function(b){
      b = ee.Number(b);
      var coll = dsesr.filter(ee.Filter.eq('bin', b));
      return coll.reduce(ee.Reducer.percentile([p])).rename('tmp');
    })
  ).toBands();

  var names = bins.map(function(b){
    return ee.String('dP').cat(ee.Number(p).format()).cat('_').cat(ee.Number(b).format('%02d'));
  });

  return pIC.rename(names);
}

// Threshold images (each has 46 bands: dPxx_00 ... dPxx_45)
var dsesrP10bin = dsesrPercentileByBin(10);
var dsesrP15bin = dsesrPercentileByBin(15);
var dsesrP20bin = dsesrPercentileByBin(20);
var dsesrP25bin = dsesrPercentileByBin(25);  // replaces/duplicates your existing one


// ===============================================
// Flash drought windows (full series, cross-year)
// ===============================================
var MIN_DSESR_LEN = 4;   // 4 dSESR steps => 4×8-day bins = 32 days

var sesrSorted  = sesrWithP20.sort('system:time_start');
var dsesrSorted = dsesrWithP40.sort('system:time_start');  // has bands: dSESR + dSESR_rapid

var N  = sesrSorted.size();
var Nd = dsesrSorted.size();   // ~ N - 1

var sesrList = sesrSorted.toList(N);
var dList    = dsesrSorted.toList(Nd);

var kList = ee.List.sequence(MIN_DSESR_LEN, N.subtract(1));

var fdWin3 = ee.ImageCollection(
  kList.map(function(k){
    k = ee.Number(k);

    // Criterion 2: final SESR below 20th percentile (bin-based)
    var sesrImg = ee.Image(sesrList.get(k));
    var finalOK = sesrImg.select('SESR_belowP20');

    // 4 dSESR steps ending at k: indices (k-4, k-3, k-2, k-1) in dList
    var d0 = ee.Image(dList.get(k.subtract(4)));
    var d1 = ee.Image(dList.get(k.subtract(3)));
    var d2 = ee.Image(dList.get(k.subtract(2)));
    var d3 = ee.Image(dList.get(k.subtract(1)));

    // Criterion 3a+3b (your choice): ≥3 of 4 are “rapid”
    var r0 = d0.select('dSESR_rapid');
    var r1 = d1.select('dSESR_rapid');
    var r2 = d2.select('dSESR_rapid');
    var r3 = d3.select('dSESR_rapid');

    var rapidSum = r0.add(r1).add(r2).add(r3);
    var rapidOK  = rapidSum.gte(3);

    // ----------------------------
    // Criterion 4 (STRICT): mean(dSESR over FD period)
    // < 25th percentile climatological dSESR for that pixel & time-of-year(bin)
    // ----------------------------
    var meanD = d0.select('dSESR')
                 .add(d1.select('dSESR'))
                 .add(d2.select('dSESR'))
                 .add(d3.select('dSESR'))
                 .divide(4);

    // per-step bin-based P25 thresholds (match each dSESR step’s bin)
    var b0 = ee.Number(d0.get('bin')).format('%02d');
    var b1 = ee.Number(d1.get('bin')).format('%02d');
    var b2 = ee.Number(d2.get('bin')).format('%02d');
    var b3 = ee.Number(d3.get('bin')).format('%02d');

    var t0 = dsesrP25bin.select(ee.String('dP25_').cat(b0));
    var t1 = dsesrP25bin.select(ee.String('dP25_').cat(b1));
    var t2 = dsesrP25bin.select(ee.String('dP25_').cat(b2));
    var t3 = dsesrP25bin.select(ee.String('dP25_').cat(b3));

    var meanT = t0.add(t1).add(t2).add(t3).divide(4);

    var crit4OK = meanD.lt(meanT);

    // final window mask
    var fdMask = finalOK.and(rapidOK).and(crit4OK)
                        .rename('FDwin3')
                        .copyProperties(sesrImg, ['system:time_start']);

    return fdMask;
  })
);

// =====================================================
// STEP 2 (Intensity): classify each qualifying FD window (FD1–FD4)
// based on mean(dSESR) vs bin-based climatological percentiles
// =====================================================

var fdWinWithIntensity = ee.ImageCollection(
  kList.map(function(k){
    k = ee.Number(k);

    // --- rebuild the same FD mask logic (must match fdWin3 exactly) ---
    var sesrImg = ee.Image(sesrList.get(k));
    var finalOK = sesrImg.select('SESR_belowP20');

    var d0 = ee.Image(dList.get(k.subtract(4)));
    var d1 = ee.Image(dList.get(k.subtract(3)));
    var d2 = ee.Image(dList.get(k.subtract(2)));
    var d3 = ee.Image(dList.get(k.subtract(1)));

    var r0 = d0.select('dSESR_rapid');
    var r1 = d1.select('dSESR_rapid');
    var r2 = d2.select('dSESR_rapid');
    var r3 = d3.select('dSESR_rapid');

    var rapidSum = r0.add(r1).add(r2).add(r3);
    var rapidOK  = rapidSum.gte(3);

    var meanD = d0.select('dSESR')
      .add(d1.select('dSESR'))
      .add(d2.select('dSESR'))
      .add(d3.select('dSESR'))
      .divide(4);

    var b0 = ee.Number(d0.get('bin')).format('%02d');
    var b1 = ee.Number(d1.get('bin')).format('%02d');
    var b2 = ee.Number(d2.get('bin')).format('%02d');
    var b3 = ee.Number(d3.get('bin')).format('%02d');

    // P25 for Criterion 4 check (same as your fdWin3)
    var t25 = dsesrP25bin.select(ee.String('dP25_').cat(b0))
      .add(dsesrP25bin.select(ee.String('dP25_').cat(b1)))
      .add(dsesrP25bin.select(ee.String('dP25_').cat(b2)))
      .add(dsesrP25bin.select(ee.String('dP25_').cat(b3)))
      .divide(4);

    var crit4OK = meanD.lt(t25);

    var fdMask = finalOK.and(rapidOK).and(crit4OK).rename('FDwin3');

    // --- intensity thresholds (mean of P10/P15/P20/P25 across the 4 steps) ---
    var t10 = dsesrP10bin.select(ee.String('dP10_').cat(b0))
      .add(dsesrP10bin.select(ee.String('dP10_').cat(b1)))
      .add(dsesrP10bin.select(ee.String('dP10_').cat(b2)))
      .add(dsesrP10bin.select(ee.String('dP10_').cat(b3)))
      .divide(4);

    var t15 = dsesrP15bin.select(ee.String('dP15_').cat(b0))
      .add(dsesrP15bin.select(ee.String('dP15_').cat(b1)))
      .add(dsesrP15bin.select(ee.String('dP15_').cat(b2)))
      .add(dsesrP15bin.select(ee.String('dP15_').cat(b3)))
      .divide(4);

    var t20 = dsesrP20bin.select(ee.String('dP20_').cat(b0))
      .add(dsesrP20bin.select(ee.String('dP20_').cat(b1)))
      .add(dsesrP20bin.select(ee.String('dP20_').cat(b2)))
      .add(dsesrP20bin.select(ee.String('dP20_').cat(b3)))
      .divide(4);

    // FD categories:
    // FD1: [P20, P25) ; FD2: [P15, P20) ; FD3: [P10, P15) ; FD4: < P10
    var intensity = ee.Image(0)
      .where(fdMask.and(meanD.lt(t25)).and(meanD.gte(t20)), 1)
      .where(fdMask.and(meanD.lt(t20)).and(meanD.gte(t15)), 2)
      .where(fdMask.and(meanD.lt(t15)).and(meanD.gte(t10)), 3)
      .where(fdMask.and(meanD.lt(t10)), 4)
      .rename('FD_intensity');

    return fdMask
      .addBands(meanD.rename('meanD'))
      .addBands(intensity)
      .copyProperties(sesrImg, ['system:time_start']);
  })
);

      i.eq(0),

      fdList.get(i.subtract(1))
    ));
    return curr.eq(1).and(prev.eq(0)).toByte().rename('FD_start');
  })
);

function addSegmentedLegend(title, palette, minVal, maxVal, position, nTicks) {
  nTicks = nTicks || 7;

  var legend = ui.Panel({
    style: { 
      position: position, 
      padding: '8px 12px',
      backgroundColor: 'rgba(255,255,255,0.92)',
      border: '1px solid #ccc'
    }
  });

  // Title
  legend.add(ui.Label({ 
    value: title,
    style: { 
      fontWeight: 'bold', 
      fontSize: '14px', 
      margin: '0 0 6px 0',
      whiteSpace: 'nowrap'
    }
  }));

  // ── Continuous gradient bar ──
  var colorBar = ui.Thumbnail({
    image: ee.Image.pixelLonLat().select('latitude'),
    params: {
      bbox      : [0, 0, 0.1, 1],
      dimensions: '12x140',
      format    : 'png',
      min: 0, max: 1,
      palette   : palette.slice()
    },
    // CRITICAL: Set right margin to 0px so the ticks can touch it
    style: { stretch: 'vertical', margin: '0 0px 0 0', padding: '0' } 
  });

  // ── Tick marks panel ──
  var ticksPanel = ui.Panel({ 
    // CRITICAL: Set left margin to 0px to eliminate the gap
    style: { margin: '0 0 0 0px', padding: '0' } 
  });
  
  for (var i = 0; i <= nTicks; i++) {
    ticksPanel.add(ui.Label({
      value: '-',
      style: {
        fontSize: '10px',
        color   : '#333',
        // Kept your formula intact, but ensured no extra horizontal padding ruins the connection
        margin  : i === 0 ? '0 0 0 0' : (140/nTicks - 14) + 'px 0 0 0',
        padding : '0'
      }
    }));
  }

  // // ── Tick labels panel ──
  // var labelsPanel = ui.Panel({ style: { margin: '0', padding: '0' } });
  // for (var i = 0; i <= nTicks; i++) {
  //   var val = Math.round(maxVal - i * (maxVal - minVal) / nTicks);
  //   labelsPanel.add(ui.Label({
  //     value: String(val),
  //     style: {
  //       fontSize: '10px',
  //       margin  : i === 0 ? '0 0 0 4px' : (140/nTicks - 14) + 'px 0 0 4px',
  //       padding : '0',
  //       color   : '#333'
  //     }
  //   }));
  // }
  
  // ── Tick labels panel ──
  var labelsPanel = ui.Panel({ style: { margin: '0', padding: '0' } });
  for (var i = 0; i <= nTicks; i++) {
    var val = maxVal - i * (maxVal - minVal) / nTicks;
    
    // Check if the value is a decimal. If yes, keep 1 decimal place; otherwise keep as integer.
    var formattedVal = (minVal % 1 !== 0 || maxVal % 1 !== 0) ? val.toFixed(1) : Math.round(val);
    
    labelsPanel.add(ui.Label({
      value: String(formattedVal),
      style: {
        fontSize: '10px',
        margin  : i === 0 ? '0 0 0 4px' : (140/nTicks - 14) + 'px 0 0 4px',
        padding : '0',
        color   : '#333'
      }
    }));
  }

  // ── Row: bar | ticks | labels ──
  legend.add(ui.Panel({
    widgets: [
      colorBar, 
      ticksPanel,
      labelsPanel
    ],
    layout: ui.Panel.Layout.flow('horizontal'),
    style : { margin: '0', padding: '0' }
  }));

  Map.add(legend);
}

var L = fdI.toList(fdI.size());

// 2) Iterate through time, track events per pixel, and emit EVENT-END records
//    Event intensity = max FD_intensity within that contiguous event
//    Event duration  = (nWindows + (MIN_DSESR_LEN - 1)) * 8 days
//    Event start-year stored as a band (startYr) for START-year frequency

var init = ee.Dictionary({
  prevAny: ee.Image(0).toByte(),      // previous FDwin3 (0/1)
  maxInt:  ee.Image(0).toByte(),      // running max intensity in current event
  nWin:    ee.Image(0).toInt16(),     // # windows in current event
  startYr: ee.Image(0).toInt16(),     // start year of current event (per pixel)
  out:     ee.List([])                // list of sparse event-end images
});

var iter = ee.List.sequence(0, fdI.size().subtract(1)).iterate(function(i, state){
  state = ee.Dictionary(state);
  i = ee.Number(i);

  var img   = ee.Image(L.get(i));
  var any   = img.select('FDwin3').unmask(0).toByte();         // 0/1
  var inten = img.select('FD_intensity').unmask(0).toByte();   // 0..4

  var yr = ee.Number(ee.Date(img.get('system:time_start')).get('year'));

  var prevAny = ee.Image(state.get('prevAny'));
  var maxInt  = ee.Image(state.get('maxInt'));
  var nWin    = ee.Image(state.get('nWin'));
  var startYr = ee.Image(state.get('startYr'));
  var outList = ee.List(state.get('out'));

  var start = any.eq(1).and(prevAny.eq(0)).toByte();  // event starts this step
  var end   = any.eq(0).and(prevAny.eq(1)).toByte();  // event ends at previous step

  // Update running state when inside event
  var maxInt_new = ee.Image(0).toByte()
    .where(start.eq(1), inten)                               // start: reset to this window intensity
    .where(any.eq(1).and(start.eq(0)), maxInt.max(inten));   // continue: max over event

  var nWin_new = ee.Image(0).toInt16()
    .where(start.eq(1), 1)
    .where(any.eq(1).and(start.eq(0)), nWin.add(1));

  var startYr_new = ee.Image(0).toInt16()
    .where(start.eq(1), yr)
    .where(any.eq(1).and(start.eq(0)), startYr);

  // Emit EVENT-END record (sparse) using STATE FROM PREVIOUS STEP:
  // maxInt & nWin & startYr correspond to the event that just ended.
  var durDays = nWin.add(MIN_DSESR_LEN - 1).multiply(8).toFloat();

  var e1 = end.and(maxInt.eq(1)).rename('e1').toByte();
  var e2 = end.and(maxInt.eq(2)).rename('e2').toByte();
  var e3 = end.and(maxInt.eq(3)).rename('e3').toByte();
  var e4 = end.and(maxInt.eq(4)).rename('e4').toByte();

  var d1 = durDays.updateMask(e1).rename('d1');
  var d2 = durDays.updateMask(e2).rename('d2');
  var d3 = durDays.updateMask(e3).rename('d3');
  var d4 = durDays.updateMask(e4).rename('d4');

  // var eventEndRecord = e1.addBands([e2,e3,e4,d1,d2,d3,d4, startYr.rename('startYr').toInt16()])
  //   .copyProperties(img, ['system:time_start']); // timestamp = current step (first non-event step)
    
  var eventEndRecord = e1.addBands([e2,e3,e4,d1,d2,d3,d4, startYr.rename('startYr').toInt16()])
    .setDefaultProjection(proj)
    .reproject({crs: proj.crs(), scale: scale})
    .copyProperties(img, ['system:time_start']);
    

  outList = outList.add(eventEndRecord);

  // Advance state (reset outside events)
  return ee.Dictionary({
    prevAny: any,
    maxInt:  ee.Image(0).toByte().where(any.eq(1), maxInt_new),
    nWin:    ee.Image(0).toInt16().where(any.eq(1), nWin_new),
    startYr: ee.Image(0).toInt16().where(any.eq(1), startYr_new),
    out:     outList
  });
}, init);

// =====================================================
// STEP 4B) REVISED EVENT FREQUENCY (% of years with an event peaking at FDx)
// Safe, optimized implementation preventing zero-band and memory errors
// =====================================================
function freqByLevel_events(level) {
  var targetBand = 'e' + level; 
  
  // Isolate the target intensity and map calendar years to the images
  var eventsWithYear = outIC.select(targetBand).map(function(img) {
    var y = ee.Date(img.get('system:time_start')).get('year');
    return img.rename('occurred').set('year', y);
  });
  
  // Calculate annual occurrence using a safe collection-merge structure
  var annualOccurrence = ee.ImageCollection(
    years.map(function(y) {
      y = ee.Number(y);
      var yearCollection = eventsWithYear.filter(ee.Filter.eq('year', y));
      
      // Bulletproof dummy layer prevents zero-band crash if a year has no events
      var dummyBlank = ee.Image(0).rename(['occurred']).toByte();
      var hasEvent = yearCollection.merge(ee.ImageCollection([dummyBlank])).max();
      
      return hasEvent.rename('anyEvent');
    })
  );
  
  // Sum up years matching the condition, convert to percentage of total study timeline
  return annualOccurrence.sum()
                         .divide(totalYears)
                         .multiply(100)
                         .rename('FD' + level + '_freqPct');
}
